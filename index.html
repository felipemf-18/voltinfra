import os
import requests

url = "https://api.voltinfra.xyz/v1/rerank"
headers = {
    "Content-Type": "application/json",
    "Authorization": f"Bearer {os.environ.get('VOLT_API_KEY')}"
}

payload = {
    "model": "qwen/qwen3-reranker-8b",
    "query": "What is the memory footprint of an 8B INT8 model?",
    "documents": [
        "An 8B parameter model requires approximately 8.2 GB of VRAM in INT8 quantization.",
        "The RTX 3060 features 12 GB of GDDR6 memory with 360 GB/s bandwidth.",
        "VoltInfra provides low-latency vector reranking APIs for enterprise search."
    ],
    "top_n": 3,
    "return_documents": False
}

response = requests.post(url, json=payload, headers=headers)
data = response.json()

for hit in data.get("results", []):
    print(f"Doc #{hit['index']} | Relevance Score: {hit['relevance_score']:.4f}")

Data Privacy & Sovereignty

Zero Data Retention (ZDR)

Search queries and candidate corpus passages are processed strictly in volatile GPU tensor memory. Zero logging to disk, zero IP tracking, and zero model training on customer payloads.

Ephemeral Scoring Pass
Query-document tokens exist strictly in GPU forward-pass activations and are flushed immediately upon returning scores.
Zero Disk Storage
Runtime containers operate with disk caching and payload logging permanently disabled at the engine level.
No Model Training
Customer traffic is never collected, pooled, or utilized for model fine-tuning, training datasets, or benchmarking.
